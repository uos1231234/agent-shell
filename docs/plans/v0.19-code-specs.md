```typescript
import type { ToolRegistry } from '../../shell/registry.js'

export type InjectionContext = {
  conversationHistory: readonly unknown[]
  registry: ToolRegistry
  sessionId?: string
  agentId?: string
  round: number
}

export type ContextInjectionSource = {
  name: string
  priority: number
  inject: (ctx: InjectionContext) => Promise<string | null>
}

export class ContextInjector {
  private readonly sources: ContextInjectionSource[] = []

  register(source: ContextInjectionSource): void {
    this.sources.push(source)
    this.sources.sort((a, b) => a.priority - b.priority)
  }

  async inject(ctx: InjectionContext): Promise<string[]> {
    const messages: string[] = []
    for (const source of this.sources) {
      const content = await source.inject(ctx)
      if (content != null && content.length > 0) messages.push(content)
    }
    return messages
  }

  async clear(): Promise<void> {
    this.sources.length = 0
  }
}
```

### 3.1 `src/im/prompt/section-builder.ts`

```typescript
import type { PromptMode } from './types.js'
import { FULL_PROMPT_TEMPLATE, MINIMAL_PROMPT_TEMPLATE, NONE_PROMPT_TEMPLATE } from './static-prompt.js'
import type { ToolRegistry } from '../../shell/registry.js'

export function buildStaticPrompt(opts: {
  mode: PromptMode
  agentName: string
  registry: ToolRegistry
  toolingContent?: string  // 可选覆盖，不传则自动生成
}): string {
  let template: string
  switch (opts.mode) {
    case 'full': template = FULL_PROMPT_TEMPLATE; break
    case 'minimal': template = MINIMAL_PROMPT_TEMPLATE; break
    case 'none': template = NONE_PROMPT_TEMPLATE; break
  }

  let result = template.replace(/\{agent_name\}/g, opts.agentName)

  // 替换 tooling_section
  if (result.includes('{tooling_section}')) {
    const tooling = opts.toolingContent ?? buildToolingSection(opts.registry, opts.mode)
    result = result.replace('{tooling_section}', tooling)
  }

  // dynamic_sections 留给 ContextInjector 在运行时填充
  // 这里只移除占位符（如果不需要动态注入的话）
  if (!result.includes('{dynamic_sections}')) return result
  // dynamic_sections 保留原样，由 loop.ts 在运行时替换
  return result
}

function buildToolingSection(registry: ToolRegistry, mode: PromptMode): string {
  if (mode === 'minimal' || mode === 'none') return ''

  const toolNames = registry.listSystemToolNames()
  const mcpMetas = registry.listMCPServerMetas()
  const skillMetas = registry.listLoadableSkillMetas()

  const lines = ['你可以使用以下工具完成任务：']
  for (const name of toolNames) {
    const tool = registry.getSystemTool(name)
    if (tool) lines.push(`- ${name}：${tool.description}`)
  }
  if (mcpMetas.size > 0 || skillMetas.size > 0) {
    lines.push('')
    lines.push('使用 load_tools 工具按需加载 MCP server 和 skill 的工具定义。')
    lines.push('可用的 MCP server 和 skill 列表见动态注入的上下文。')
  }
  return lines.join('\n')
}
```

### 4.1 `src/im/prompt/layer-builder.ts`

```typescript
import type { PromptLayer } from './types.js'

/**
 * 按 priority 排序并合并多层配置。
 * priority 高的在后面（后面的覆盖前面的同名内容）。
 * 返回合并后的完整字符串。
 */
export function buildLayeredPrompt(layers: PromptLayer[]): string {
  if (layers.length === 0) return ''
  const sorted = [...layers].sort((a, b) => a.priority - b.priority)
  return sorted.map(l => l.content).filter(c => c.length > 0).join('\n\n')
}
```

### 4.2 `src/im/prompt/file-loader.ts`

```typescript
import { readFile } from 'node:fs/promises'
import { join } from 'node:path'
import type { PromptLayer, PromptLayerName } from './types.js'

const MAX_LAYER_BYTES = 20 * 1024  // 20KB per file

export async function loadPromptLayers(opts: {
  basePath: string
  projectFile?: string
  localFile?: string
  globalPath?: string
  userPath?: string
}): Promise<PromptLayer[]> {
  const layers: PromptLayer[] = []
  const projectFile = opts.projectFile ?? 'AGENTS.md'
  const localFile = opts.localFile ?? '.claude/local.md'

  // 加载顺序：managed → user → project → local（priority 递增）
  if (opts.globalPath) {
    const content = await safeRead(opts.globalPath)
    if (content) layers.push({ name: 'managed', content, priority: 0 })
  }

  const userPath = opts.userPath ?? join(process.env.HOME ?? process.env.USERPROFILE ?? '', '.agent-shell', 'PROMPT.md')
  const userContent = await safeRead(userPath)
  if (userContent) layers.push({ name: 'user', content: userContent, priority: 10 })

  const projectPath = join(opts.basePath, projectFile)
  const projectContent = await safeRead(projectPath)
  if (projectContent) layers.push({ name: 'project', content: projectContent, priority: 20 })

  const localPath = join(opts.basePath, localFile)
  const localContent = await safeRead(localPath)
  if (localContent) layers.push({ name: 'local', content: localContent, priority: 30 })

  return layers
}

async function safeRead(path: string): Promise<string | null> {
  try {
    const content = await readFile(path, 'utf-8')
    if (content.length > MAX_LAYER_BYTES) {
      console.warn(`[prompt-loader] ${path} exceeds ${MAX_LAYER_BYTES} bytes, truncating`)
      return content.slice(0, MAX_LAYER_BYTES)
    }
    return content.length > 0 ? content : null
  } catch {
    return null  // 文件不存在时静默跳过
  }
}
```

```typescript
import type { ContextInjectionSource } from '../context-injection.js'
import { isDynamicToolSchemaMessage, type DynamicToolSchemaMessage } from '../../dynamic-tool-context.js'
import type { OpenAITool } from '../../../protocol/types.js'

export function createDynamicSchemaInjection(): ContextInjectionSource {
  return {
    name: 'loaded_dynamic_tools',
    priority: 40,
    inject: async (ctx) => {
      const tools: OpenAITool[] = []
      const sources: string[] = []

      for (const msg of ctx.conversationHistory) {
        if (!isDynamicToolSchemaMessage(msg)) continue
        const m = msg as DynamicToolSchemaMessage

          : m.origin.source.name
        sources.push(label)

      if (tools.length === 0) return null
      const toolNames = tools.map(t => t.function.name).join(', ')
      return `[已加载工具] 来源: ${sources.join(', ')} | 工具: ${toolNames}`
    },
  }
```

## 6. Auto Memory 完整代码

### 6.1 `src/im/memory/auto-memory.ts`

```typescript
import { readFile, writeFile, mkdir } from 'node:fs/promises'
import { join } from 'node:path'
import type { MemoryNote, MemoryNoteType, MemoryIndex } from './types.js'

const INDEX_FILE = 'MEMORY.md'
const NOTE_FILES: Record<MemoryNoteType, string> = {
  user: 'user.md',
  feedback: 'feedback.md',
  project: 'project.md',
  reference: 'reference.md',
}

export class AutoMemory {
  private readonly dir: string

  constructor(opts: { basePath: string; projectId: string }) {
    this.dir = join(opts.basePath, '.agent-shell', 'memory', opts.projectId)
  }

  async readIndex(): Promise<MemoryIndex> {
    try {
      const raw = await readFile(join(this.dir, INDEX_FILE), 'utf-8')
      const notes = this.parseNotes(raw)
      return { notes, lastUpdated: new Date().toISOString() }
    } catch {
      return { notes: [], lastUpdated: new Date().toISOString() }
    }
  }

  async readNotes(type?: MemoryNoteType): Promise<MemoryNote[]> {
    if (type) {
      return this.readNoteFile(type)
    }
    const all: MemoryNote[] = []
    for (const t of Object.keys(NOTE_FILES) as MemoryNoteType[]) {
      all.push(...await this.readNoteFile(t))
    }
    return all
  }

  async writeNote(note: MemoryNote): Promise<void> {
    await mkdir(this.dir, { recursive: true })
    const fileName = NOTE_FILES[note.type]
    const filePath = join(this.dir, fileName)
    const line = `[${note.timestamp}] ${note.content}${note.source ? ` (source: ${note.source})` : ''}\n`
    let existing = ''
    try { existing = await readFile(filePath, 'utf-8') } catch { /* new file */ }
    await writeFile(filePath, existing + line, 'utf-8')
  }

  async searchNotes(query: string): Promise<MemoryNote[]> {
    const all = await this.readNotes()
    const lower = query.toLowerCase()
    return all.filter(n => n.content.toLowerCase().includes(lower))
  }

  async close(): Promise<void> {
    // 预留：未来可加锁释放
  }

  private async readNoteFile(type: MemoryNoteType): Promise<MemoryNote[]> {
    const filePath = join(this.dir, NOTE_FILES[type])
    try {
      const raw = await readFile(filePath, 'utf-8')
      return this.parseNotes(raw, type)
    } catch {
      return []
    }
  }

  private parseNotes(raw: string, defaultType?: MemoryNoteType): MemoryNote[] {
    const notes: MemoryNote[] = []
    for (const line of raw.split('\n')) {
      const match = line.match(/^\[(.+?)\]\s+(.+?)(?:\s+\(source:\s+(.+?)\))?$/)
      if (!match) continue
      notes.push({
        type: (match[3] ? this.inferType(match[3]) : defaultType) ?? 'project',
        content: match[2],
        timestamp: match[1],
        source: match[3],
      })
    }
    return notes
  }

  private inferType(source: string): MemoryNoteType | undefined {
    for (const [type, fileName] of Object.entries(NOTE_FILES)) {
      if (source.includes(fileName)) return type as MemoryNoteType
    }
    return undefined
  }
}
```

## 7. HookSystem 完整代码

### 7.1 `src/im/hooks/hook-system.ts`

```typescript
import type { HookEvent, HookHandler, HookContext } from './types.js'

export class HookSystem {
  private readonly handlers: HookHandler[] = []

  register(handler: HookHandler): void {
    this.handlers.push(handler)
  }

  async emit(event: Exclude<HookEvent, 'PreToolUse'>, ctx: HookContext): Promise<void> {
    for (const h of this.handlers) {
      if (h.event === event) {
        await h.handler(ctx)
      }
    }
  }

  async emitPreToolUse(ctx: HookContext): Promise<boolean> {
    // PreToolUse: 任一 handler 返回 false 则阻止
    for (const h of this.handlers) {
      if (h.event === 'PreToolUse') {
        const result = await h.handler(ctx)
        if (result === false) return false
      }
    }
    return true
  }

  async clear(): Promise<void> {
    this.handlers.length = 0
  }
}
```

```typescript
const CONSECUTIVE_ERROR_THRESHOLD = 3

export function createErrorRecoveryHook(): HookHandler & { shouldSuggestAlternatives(): boolean } {
  let consecutiveErrors = 0

  return {
    event: 'PostToolUse',
    handler: async (ctx) => {
      if (ctx.error) {
        consecutiveErrors++
      } else {
        consecutiveErrors = 0
      }
    },
    shouldSuggestAlternatives: () => consecutiveErrors >= CONSECUTIVE_ERROR_THRESHOLD,
  }
}
```
